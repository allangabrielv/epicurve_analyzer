%========================= 5 · TRATAMENTO ==============================
\begin{frame}{tratamento dos dados}
  \colab
  \begin{enumerate}
    \item Ler só \dest{6 de 18 colunas} — eficiência de memória com dado grande.
    \item Filtrar o \dest{nível estadual de SP} (já é a soma diária de todos os municípios).
    \item Garantir \dest{frequência diária contínua} (sem buracos no calendário).
    \item \dest{Zerar valores negativos} das correções retroativas do governo.
    \item Converter tipos e ordenar $\rightarrow$ \bom{série diária de 762 dias}.
  \end{enumerate}
  \vspace{0.3em}
  \begin{tcolorbox}[pitchbox]
    \footnotesize \textbf{Honestidade:} para SP a série já vinha íntegra —
    \bom{0 datas faltando e 0 negativos}. As travas existem (código robusto para qualquer
    estado), mas nada precisou ser corrigido.
  \end{tcolorbox}
\end{frame}

%========================= 14 · CONCLUSÕES ============================
\begin{frame}{conclusões}
  \begin{enumerate}
    \item \dest{É um problema de série temporal.} Decomposição, ADF/KPSS e ACF/PACF
          revelaram a sazonalidade semanal (período 7) da notificação.
    \item \dest{Quem capta a sazonalidade vence.} Holt-Winters e SARIMA batem folgado o
          ARIMA (linha plana) e o naïve.
    \item \dest{LazyPredict é o atalho.} Em uma linha, testou dezenas de modelos e chegou
          perto dos clássicos. Automatizar acha o candidato; \emph{conhecer a série} dá o melhor resultado.
  \end{enumerate}
