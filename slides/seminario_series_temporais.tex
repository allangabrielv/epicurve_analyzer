\section{Introdução aos modelos de séries temporais}
%=======================================================================

\begin{frame}{O que é uma série temporal?}
  \begin{tcolorbox}[pitchtitled,title=Definição]
    Uma \dest{série temporal} é uma sequência de observações de uma variável
    indexadas no tempo e \dest{ordenadas cronologicamente}:
    \eqbox{\{y_t\}_{t=1}^{n} = y_1, y_2, \ldots, y_n .}
  \end{tcolorbox}
  \vspace{0.4em}
  A característica essencial é a \dest{dependência temporal}: observações
  próximas tendem a ser correlacionadas — o que \dest{viola} a hipótese de
  independência da estatística clássica.
  \begin{itemize}
    \item \textbf{Exemplos:} casos diários de uma doença, PIB, temperatura, ações.
    \item \textbf{Objetivos:} descrever, explicar a estrutura e, sobretudo,
          \dest{prever} ($\hat{y}_{n+h}$).
  \end{itemize}
\end{frame}

\begin{frame}{Componentes de uma série temporal}
  \eqbox{y_t = \underbrace{T_t}_{\text{tend\^encia}} + \underbrace{S_t}_{\text{sazonal}}
        + \underbrace{C_t}_{\text{ciclo}} + \underbrace{\varepsilon_t}_{\text{ru\'ido}}
        \quad\text{(aditiva)};\qquad y_t = T_t\cdot S_t\cdot\varepsilon_t\ \text{(multiplicativa)}}
  \begin{columns}[T]
    \column{0.5\textwidth}
      \begin{tcolorbox}[pitchbox]
        \textbf{Tendência $T_t$}: movimento de longo prazo.\\[0.3em]
        \textbf{Sazonalidade $S_t$}: padrão que se repete em período fixo $s$.
      \end{tcolorbox}
    \column{0.5\textwidth}
      \begin{tcolorbox}[pitchbox]
        \textbf{Ciclo $C_t$}: oscilações de período não fixo (ondas).\\[0.3em]
        \textbf{Irregular $\varepsilon_t$}: ruído aleatório.
      \end{tcolorbox}
  \end{columns}
  \vspace{0.3em}
  \centering\footnotesize A forma multiplicativa vira aditiva no \dest{logaritmo}:
  $\log y_t=\log T_t+\log S_t+\log\varepsilon_t$.
\end{frame}

\begin{frame}{Três abordagens — o fio condutor}
  \begin{enumerate}
    \item \textbf{Decomposição / Regressão} — modela tendência e sazonalidade
          explicitamente. \emph{Partimos da regressão linear simples.}
    \item \textbf{Alisamento exponencial} — médias com pesos que decaem (SES,
          Holt, Holt-Winters).
    \item \textbf{Box--Jenkins (ARIMA)} — modela diretamente a
          \dest{autocorrelação}: AR, MA, ARMA, ARIMA, SARIMA, SARIMAX.
  \end{enumerate}
  \begin{tcolorbox}[alertbox]
    \textbf{Estudo de caso:} aplicaremos \dest{todos} esses modelos aos
    \dest{casos diários de COVID-19 em São Paulo}.
  \end{tcolorbox}
\end{frame}

\begin{frame}{Estudo de caso: COVID-19 em São Paulo}
  \begin{columns}[T]
    \column{0.42\textwidth}
      \begin{tcolorbox}[pitchbox]
        \footnotesize
        \textbf{Fonte:} DataSUS / Brasil.IO (\texttt{caso\_full.csv.gz}).
        \begin{itemize}\footnotesize
          \item $\approx 3{,}85$ mi de registros municipais.
          \item Recorte: \dest{estado de SP} (agregado).
          \item \dest{25/02/2020 a 27/03/2022}; $n=762$ dias.
          \item $5{,}23$ mi de casos; $167$ mil óbitos.
          \item Pico: \dest{37.611 casos} (03/02/2022).
        \end{itemize}
        Série diária $\Rightarrow$ \dest{sazonalidade semanal} ($s=7$).
      \end{tcolorbox}
    \column{0.58\textwidth}

  \end{columns}
\end{frame}

    \column{0.54\textwidth}

        \footnotesize Média, variância e autocovariâncias \dest{não dependem do tempo}:
        \[\mathbb{E}[y_t]=\mu,\ \operatorname{Var}(y_t)=\sigma^2,\ \operatorname{Cov}(y_t,y_{t+k})=\gamma_k.\]

      \begin{itemize}\footnotesize
        \item \textbf{ADF}: $H_0=$ raiz unitária (não estacionária).

        (a sazonalidade confunde os testes).\\[0.4em]
        Após $1^{\mathrm a}$ diferença: ADF $p<0{,}001$
        \bom{$\Rightarrow$ estacionária}.
      \end{tcolorbox}
  \end{columns}
\end{frame}

\section{Regressão linear simples}
%=======================================================================

\begin{frame}{Regressão linear simples: o modelo de tendência}
  Ajustamos uma \dest{reta} à série em função do tempo:
  \eqbox{y_t=\beta_0+\beta_1\,t+\varepsilon_t}
  \begin{columns}[T]
    \column{0.5\textwidth}
      \begin{tcolorbox}[pitchbox]
        \footnotesize Estimação por \dest{mínimos quadrados} (MQO):
        \[\hat\beta_1=\frac{\sum_t(t-\bar t)(y_t-\bar y)}{\sum_t(t-\bar t)^2},\quad
          \hat\beta_0=\bar y-\hat\beta_1\bar t\]
      \end{tcolorbox}
    \column{0.5\textwidth}
      \begin{tcolorbox}[pitchbox]
        \footnotesize Qualidade do ajuste:
        \[R^2=1-\frac{\sum_t(y_t-\hat y_t)^2}{\sum_t(y_t-\bar y)^2}\in[0,1]\]
      \end{tcolorbox}
  \end{columns}
  \centering\footnotesize\color{clight}
  Mesma base da disciplina de origem: \texttt{scipy.stats.linregress} e \texttt{statsmodels.OLS}.
\end{frame}

\begin{frame}{Regressão linear: tendência de casos em SP}
  \begin{columns}[T]
    \column{0.62\textwidth}
      \figura{05_regressao_linear.png}
    \column{0.38\textwidth}
      \begin{tcolorbox}[pitchbox]
        \footnotesize Janela 01/11/2020--31/03/2021 ($n=151$):
        \[\widehat{\text{casos}}=3469+\dest{73{,}3}\,t\]
        \begin{itemize}\footnotesize
          \item Tendência \dest{$+73$ casos/dia} ($p<10^{-12}$).
          \item IC 95\%: $[55{,}3;\,91{,}2]$.
          \item $R^2=\dest{0{,}30}$ (diário).
          \item $R^2=\bom{0{,}77}$ (média móvel 7d).
        \end{itemize}
      \end{tcolorbox}
  \end{columns}
\end{frame}

\begin{frame}{Por que a regressão simples não basta}
  \begin{columns}[T]
    \column{0.54\textwidth}
      O MQO supõe erros \dest{independentes} — o que quase nunca vale em séries
      temporais. A estatística de \dest{Durbin--Watson}:
      \eqbox{d=\frac{\sum_{t=2}^{n}(e_t-e_{t-1})^2}{\sum_t e_t^2}\approx 2(1-\hat\rho_1)}
      $d\approx2$: sem autocorrelação. Aqui $d=\dest{1{,}40}$.
    \column{0.46\textwidth}
      \begin{tcolorbox}[alertbox]
        \textbf{Consequência}\\
        Resíduos \dest{autocorrelacionados} (padrão semanal) $\Rightarrow$
        erros-padrão subestimados e previsões pobres.
      \end{tcolorbox}
      \begin{tcolorbox}[pitchbox]
        \textbf{Conclusão:} precisamos de modelos que tratem a
        \dest{dependência temporal}. \emph{É o resto do seminário.}
      \end{tcolorbox}
  \end{columns}
\end{frame}

\begin{frame}{Ponte com Cálculo Numérico: linearização}
  \begin{columns}[T]
    \column{0.6\textwidth}
      \figura{06_linearizacao_exponencial.png}
    \column{0.4\textwidth}
      \begin{tcolorbox}[pitchbox]
        \footnotesize No início de uma onda o crescimento é \dest{exponencial},
        $y=A\,e^{kt}$. O logaritmo \dest{lineariza}:
        \[\ln y=\ln A+k\,t\]
        e aplicamos regressão linear em $(t,\ln y)$.\\[0.4em]
        Ômicron: $k=0{,}174$/dia, $R^2=0{,}59$, duplicação $\approx\dest{4{,}0}$ dias.
      \end{tcolorbox}
  \end{columns}
\end{frame}

\section{Alisamento exponencial}
%=======================================================================

\begin{frame}{Alisamento exponencial simples (SES)}
  A previsão é uma \dest{média ponderada} do passado, com pesos que
  \dest{decaem exponencialmente}:
  \eqbox{\hat y_{t+1}=\alpha\,y_t+(1-\alpha)\hat y_t,\qquad 0<\alpha<1}
  Forma de componente de nível: $\ \ell_t=\alpha y_t+(1-\alpha)\ell_{t-1},\quad \hat y_{t+h}=\ell_t$.
  \begin{itemize}
    \item $\alpha\to1$: reage rápido; $\alpha\to0$: muito suave.
    \item Adequado a séries \dest{sem tendência nem sazonalidade} — precisamos estender.
  \end{itemize}
\end{frame}

\begin{frame}{Holt e Holt--Winters}
  \begin{tcolorbox}[pitchtitled,title=Holt (tendência)]
    \footnotesize
    \[\ell_t=\alpha y_t+(1-\alpha)(\ell_{t-1}+b_{t-1}),\quad
      b_t=\beta(\ell_t-\ell_{t-1})+(1-\beta)b_{t-1},\quad \hat y_{t+h}=\ell_t+h\,b_t\]
  \end{tcolorbox}
  \begin{tcolorbox}[pitchtitled,title={Holt--Winters (tendência + sazonalidade $s_t$, período $m$)}]
    \footnotesize
    \[s_t=\gamma(y_t-\ell_{t-1}-b_{t-1})+(1-\gamma)s_{t-m},\quad
      \hat y_{t+h}=\ell_t+hb_t+s_{t+h-m(\lfloor(h-1)/m\rfloor+1)}\]
  \end{tcolorbox}
  \centering\footnotesize\color{clight}
  Usamos \dest{tendência amortecida} e modelamos em $\log(1+\text{casos})$.
\end{frame}

\begin{frame}{Alisamento exponencial: previsão de 28 dias}
  \begin{columns}[T]
    \column{0.6\textwidth}
      \figura{07_alisamento_exponencial.png}
    \column{0.4\textwidth}
      \begin{tcolorbox}[pitchbox]
        \footnotesize
        \begin{tabular}{lcc}
          \textbf{Modelo} & $\alpha$ & \textbf{RMSE}\\[0.2em]
          SES & 0,23 & 725\\
          Holt & 0,14 & 709\\
          \bom{Holt--Winters} & 0,24 & \bom{498}\\
        \end{tabular}\\[0.6em]
        Só o \dest{Holt--Winters} capta o padrão semanal — reduz o erro em
        $\sim$30\%. A sazonalidade é a estrutura \dest{dominante}.
      \end{tcolorbox}
  \end{columns}
\end{frame}

\section{Autocorrelação: ACF e PACF}
%=======================================================================

\begin{frame}{Ruído branco, autocovariância e ACF}
  \begin{tcolorbox}[pitchtitled,title=Ruído branco]
    \footnotesize $\varepsilon_t$ i.i.d., média 0, variância $\sigma^2$, sem
    autocorrelação — o "tijolo" dos modelos.
  \end{tcolorbox}
  \begin{columns}[T]
    \column{0.5\textwidth}
      \textbf{ACF} — correlação com a defasagem $k$:
      \eqbox{\rho_k=\frac{\gamma_k}{\gamma_0}=\frac{\operatorname{Cov}(y_t,y_{t-k})}{\operatorname{Var}(y_t)}}
    \column{0.5\textwidth}
      \textbf{PACF} $\phi_{kk}$ — correlação entre $y_t$ e $y_{t-k}$ \dest{removido}
      o efeito das defasagens intermediárias.
  \end{columns}
  \centering\footnotesize\color{clight}
  O \dest{correlograma} (ACF/PACF) é a principal ferramenta de identificação.
\end{frame}

\begin{frame}{ACF e PACF da série de SP}
  \figalt{0.72\textheight}{09_acf_pacf.png}
  \centering\scriptsize\color{clight}
  Em nível: ACF decai \dest{lentamente} (não estacionária). Na $1^{\mathrm a}$
  diferença: picos em \dest{7, 14, 21} $\Rightarrow$ \dest{sazonalidade semanal}.
\end{frame}

\section{Modelos de médias móveis (MA)}
%=======================================================================

\begin{frame}{Filtro de média móvel \texorpdfstring{$\neq$}{vs} modelo MA(q)}
  \begin{columns}[T]
    \column{0.58\textwidth}
      \figura{08_media_movel_filtro.png}
    \column{0.42\textwidth}
      \begin{tcolorbox}[alertbox]
        \footnotesize\textbf{Cuidado com a ambiguidade!}
      \end{tcolorbox}
      \begin{tcolorbox}[pitchbox]
        \footnotesize
        \begin{itemize}\footnotesize
          \item \textbf{Filtro} de média móvel: técnica \emph{descritiva} de
                suavização (remove o ruído semanal).
          \item \textbf{Modelo} MA($q$): processo \emph{estocástico} baseado em
                \dest{choques} passados.
        \end{itemize}
      \end{tcolorbox}
  \end{columns}
\end{frame}

\begin{frame}{Modelo de médias móveis MA(q)}
  \eqbox{y_t=\mu+\varepsilon_t+\theta_1\varepsilon_{t-1}+\cdots+\theta_q\varepsilon_{t-q}
        =\mu+\theta(\B)\,\varepsilon_t}
  $\B$ é o \dest{operador de retardo} ($\B y_t=y_{t-1}$);
  $\theta(\B)=1+\theta_1\B+\cdots+\theta_q\B^q$.
  \begin{itemize}
    \item Combinação linear de \dest{choques aleatórios} — "memória" de tamanho $q$.
    \item É \dest{sempre estacionário}.
    \item \dest{Identificação:} a \textbf{ACF corta} após a defasagem $q$;
          a PACF decai gradualmente.
  \end{itemize}
\end{frame}

%=======================================================================
\section{Modelos autorregressivos (AR)}
%=======================================================================

\begin{frame}{Modelo autorregressivo AR(p)}
  $y_t$ é regredido sobre seus \dest{próprios valores passados}:
  \eqbox{y_t=c+\phi_1 y_{t-1}+\cdots+\phi_p y_{t-p}+\varepsilon_t
        \;\Longleftrightarrow\; \phi(\B)\,y_t=c+\varepsilon_t}
  \begin{itemize}
    \item \textbf{Estacionariedade:} raízes de $\phi(\B)=0$ \dest{fora do círculo
          unitário} (AR(1): $|\phi_1|<1$).
    \item \dest{Identificação} (simétrica ao MA): a \textbf{PACF corta} após $p$;
          a ACF decai gradualmente.
  \end{itemize}
  \begin{tcolorbox}[pitchbox]
    \footnotesize\textbf{Dualidade:} AR($p$) estacionário $=$ MA($\infty$); MA($q$)
    invertível $=$ AR($\infty$). Por isso combinamos os dois.
  \end{tcolorbox}
\end{frame}

%=======================================================================
\section{Modelos ARMA}
%=======================================================================

\begin{frame}{ARMA(p,q): juntando AR e MA}
  \eqbox{\phi(\B)\,y_t=c+\theta(\B)\,\varepsilon_t}
  \begin{columns}[T]
    \column{0.55\textwidth}
      \footnotesize\textbf{Identificação (Box--Jenkins):}
      \begin{itemize}\footnotesize
        \item ACF e PACF \dest{ambas decaem} $\Rightarrow$ ARMA.
        \item Ordens $(p,q)$ por \dest{parcimônia} e critérios de informação:
      \end{itemize}
      \[\text{AIC}=-2\ln L+2k,\quad \text{BIC}=-2\ln L+k\ln n\]
    \column{0.45\textwidth}
      \begin{tcolorbox}[pitchbox]
        \footnotesize \textbf{Ajustes ilustrativos} (série diferenciada):
        \begin{tabular}{lc}
          \textbf{Modelo} & \textbf{AIC}\\[0.2em]
          \bom{AR(7)} & \bom{2373}\\
          MA(7) & 2407\\
          ARMA(2,2) & 2439\\
        \end{tabular}\\[0.4em]
        A componente em \emph{lag 7} pede estrutura \dest{sazonal} (SARIMA).
      \end{tcolorbox}
  \end{columns}
\end{frame}

%=======================================================================
\section{Modelos ARIMA}
%=======================================================================

\begin{frame}{Do ARMA ao ARIMA: a integração (I)}
  Quando a série \dest{não é estacionária}, aplicamos a \dest{diferenciação}
  $\Delta y_t=y_t-y_{t-1}=(1-\B)y_t$. Diferenciando $d$ vezes:
  \eqbox{\phi(\B)\,(1-\B)^d\,y_t=c+\theta(\B)\,\varepsilon_t}
  \begin{itemize}
    \item $d$ = nº de diferenças para estacionarizar (em geral 1 ou 2).
    \item AR\dest{I}MA = ARMA sobre a série \dest{diferenciada}.
    \item Modelar $\log(1+y_t)$ \dest{estabiliza a variância} antes de diferenciar.
  \end{itemize}
\end{frame}

\begin{frame}{Metodologia de Box--Jenkins}
  \begin{enumerate}
    \item \dest{Identificação} — log + diferenciação ($d$); ler ACF/PACF para $(p,q)$.
    \item \dest{Estimação} — parâmetros por máxima verossimilhança.
    \item \dest{Diagnóstico} — resíduos devem ser \dest{ruído branco} (\textbf{Ljung--Box}).
    \item \dest{Previsão} — $\hat y_{n+h}$ com intervalos de confiança.
  \end{enumerate}
  \eqbox{Q_{LB}=n(n+2)\sum_{k=1}^{m}\frac{\hat\rho_k^2}{n-k}\ \sim\ \chi^2_{m-(p+q)}
        \quad(H_0:\text{res\'iduos n\~ao correlacionados})}
\end{frame}

\begin{frame}{Diagnóstico de resíduos do modelo ajustado}
  \begin{columns}[T]
    \column{0.62\textwidth}
      \figura{11_diagnostico_residuos.png}
    \column{0.38\textwidth}
      \begin{tcolorbox}[pitchbox]
        \footnotesize Resíduos do modelo sazonal:
        \begin{itemize}\footnotesize
          \item sem padrão temporal;
          \item ACF dentro das bandas;
          \item histograma/QQ $\approx$ normal.
        \end{itemize}
      \end{tcolorbox}
      \begin{tcolorbox}[alertbox]
        \footnotesize \textbf{Ljung--Box} (lag 14): $p=\bom{0{,}59}$
        $\Rightarrow$ \bom{ruído branco — modelo adequado}.
      \end{tcolorbox}
  \end{columns}
\end{frame}

%=======================================================================
\section{Sazonalidade: SARMA e SARIMA}
%=======================================================================

\begin{frame}{Incorporando a sazonalidade: SARMA e SARIMA}
  Adicionamos operadores \dest{sazonais} de período $s$ (aqui $s=7$):
  \eqbox{\Phi_P(\B^s)\,\phi_p(\B)\,(1-\B)^d(1-\B^s)^D\,y_t
        =\Theta_Q(\B^s)\,\theta_q(\B)\,\varepsilon_t}
  \centering\textbf{\ttfamily SARIMA$(p,d,q)(P,D,Q)_s$}
  \begin{itemize}
    \item $(p,d,q)$: parte \dest{não sazonal}; $(P,D,Q)$: parte \dest{sazonal}.
    \item $(1-\B^s)^D$: \dest{diferença sazonal} ($y_t-y_{t-7}$).
    \item \textbf{SARMA}: caso com $d=D=0$. Usamos \dest{SARIMA$(2,1,2)(1,1,1)_7$}.
  \end{itemize}
\end{frame}

\begin{frame}{ARIMA vs.\ SARIMA: o ganho da sazonalidade}
  \begin{columns}[T]
    \column{0.62\textwidth}
      \figura{10_arima_sarima_previsao.png}
    \column{0.38\textwidth}
      \begin{tcolorbox}[pitchbox]
        \footnotesize
        \begin{tabular}{lcc}
          \textbf{Modelo} & \textbf{AIC} & \textbf{RMSE}\\[0.2em]
          ARIMA$(2,1,2)$ & 1877 & 714\\
          \bom{SARIMA} & \bom{1663} & \bom{567}\\
        \end{tabular}\\[0.5em]
        O ARIMA prevê uma \dest{linha plana}; o SARIMA reproduz o
        \dest{zigue-zague semanal} ($-20\%$ no erro). Faixa = IC 95\%.
      \end{tcolorbox}
  \end{columns}
\end{frame}

\section{Variáveis exógenas: SARIMAX}
%=======================================================================

\begin{frame}{Modelagem com variáveis exógenas (SARIMAX)}
  \dest{Regressão dinâmica}: regressores externos $x_{j,t}$ + erro com estrutura SARIMA:
  \eqbox{y_t=\sum_j \beta_j\,x_{j,t}+n_t,\qquad n_t\sim\text{SARIMA}(p,d,q)(P,D,Q)_s}
  \begin{itemize}
    \item Une o \dest{poder explicativo} de covariáveis à \dest{autocorrelação} do erro.
    \item Exemplos: feriados, temperatura, mobilidade, outra série relacionada.
    \item \dest{Atenção:} para prever é preciso conhecer o \dest{caminho futuro}
          das exógenas.
  \end{itemize}
\end{frame}

\begin{frame}{Exógena 1 — feriados: quando \emph{não} ajuda}

\begin{frame}{Conclusões}
  \begin{itemize}
    \item A \dest{regressão linear} acha a tendência, mas seus resíduos
          autocorrelacionados ($d=1{,}40$) exigem modelos próprios de séries temporais.
    \item \dest{Holt--Winters} e \dest{SARIMA} capturam a sazonalidade semanal e
          foram os melhores ($-30\%$ de erro).
    \item \dest{ACF/PACF} guiam $(p,d,q)(P,D,Q)_s$; o \dest{Ljung--Box} confirma a adequação.
    \item Variáveis \dest{exógenas} (SARIMAX) ajudam só quando trazem informação
          nova — feriados vs.\ casos$\to$óbitos.
  \end{itemize}
  \begin{tcolorbox}[alertbox]
    \footnotesize\textbf{Limitações dos dados:} subnotificação, atraso de divulgação
    e revisões retroativas geram a sazonalidade semanal e dificultam previsões longas.
  \end{tcolorbox}
\end{frame}
